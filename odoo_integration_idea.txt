Here is the complete `server/odoo_client.py` module designed for **Odoo 18** (`project.task` multi-assignee `user_ids` + `parent_id` sub-tasks). It authenticates via XML-RPC using an API key, caches active `res.users` records (matching against full names, email prefixes, and configurable aliases), creates a parent task for the consolidated meeting minutes, parses the `## Action Items` Markdown table generated by Ollama, and creates individual assigned sub-tasks under that parent task.

### `server/odoo_client.py`

```python
import os
import re
import logging
import xmlrpc.client
from dataclasses import dataclass
from datetime import datetime
from difflib import SequenceMatcher
from typing import Any

logger = logging.getLogger(__name__)


@dataclass
class ActionItem:
    owner: str
    task: str
    deadline: str | None = None


class OdooMeetingClient:
    """
    Odoo 18 XML-RPC client that:
      1. Resolves meeting speaker names to Odoo `res.users` IDs (exact, alias, email-prefix, or fuzzy).
      2. Creates a parent `project.task` containing the consolidated meeting minutes.
      3. Parses the Markdown Action Items table and creates individual assigned sub-tasks
         linked via `parent_id` and `user_ids` (Odoo 15–18 Many2many assignment syntax).
    """

    def __init__(
        self,
        url: str | None = None,
        db: str | None = None,
        username: str | None = None,
        api_key: str | None = None,
        project_id: int | None = None,
        speaker_aliases: dict[str, str] | None = None,
    ) -> None:
        self.url = (url or os.getenv("ODOO_URL", "")).rstrip("/")
        self.db = db or os.getenv("ODOO_DB", "")
        self.username = username or os.getenv("ODOO_USER", "")
        self.api_key = api_key or os.getenv("ODOO_API_KEY", "")
        raw_project_id = project_id or os.getenv("ODOO_PROJECT_ID")
        self.project_id = int(raw_project_id) if raw_project_id else None

        # Optional overrides, e.g. {"Ginadomenico": "Gian Domenico", "Muhammad": "muhammad.abbasi.team@gmail.com"}
        self.speaker_aliases: dict[str, str] = {
            k.strip().lower(): v.strip()
            for k, v in (speaker_aliases or {}).items()
        }

        self._uid: int | None = None
        self._models: xmlrpc.client.ServerProxy | None = None
        self._users_cache: list[dict[str, Any]] | None = None

    @property
    def is_configured(self) -> bool:
        return bool(self.url and self.db and self.username and self.api_key)

    def _connect(self) -> tuple[int, xmlrpc.client.ServerProxy]:
        if not self.is_configured:
            raise RuntimeError("Odoo XML-RPC credentials are incomplete in environment variables.")

        if self._uid is None or self._models is None:
            common = xmlrpc.client.ServerProxy(f"{self.url}/xmlrpc/2/common", allow_none=True)
            uid = common.authenticate(self.db, self.username, self.api_key, {})
            if not uid:
                raise PermissionError(f"Odoo XML-RPC authentication failed for user '{self.username}'.")
            self._uid = int(uid)
            self._models = xmlrpc.client.ServerProxy(f"{self.url}/xmlrpc/2/object", allow_none=True)

        return self._uid, self._models

    def _execute(self, model: str, method: str, args: list[Any], kwargs: dict[str, Any] | None = None) -> Any:
        uid, models = self._connect()
        return models.execute_kw(self.db, uid, self.api_key, model, method, args, kwargs or {})

    # -------------------------------------------------------------------------
    # 1. Speaker -> res.users ID Resolution
    # -------------------------------------------------------------------------

    def fetch_active_users(self, force_refresh: bool = False) -> list[dict[str, Any]]:
        """Fetches active internal users (`share = False`) from `res.users`."""
        if self._users_cache is not None and not force_refresh:
            return self._users_cache

        users = self._execute(
            "res.users",
            "search_read",
            [[["active", "=", True], ["share", "=", False]]],
            {"fields": ["id", "name", "login", "email"]},
        )
        self._users_cache = users
        return users

    def resolve_speaker_to_user_id(self, speaker_name: str, min_similarity: float = 0.78) -> int | None:
        """
        Resolves a transcript speaker label (e.g., 'Gabriele', 'Ginadomenico') to an Odoo `res.users` ID.
        Matching order:
          1. Explicit alias lookup (`speaker_aliases`)
          2. Exact case-insensitive match on `name`, `login`, or `email`
          3. First-name or email local-part match (e.g., 'gabriele' -> 'Gabriele Rossi' or 'gabriele@...')
          4. Fuzzy SequenceMatcher ratio >= `min_similarity` (handles minor STT spelling variations)
        """
        cleaned = re.sub(r"[\[\]\*@]", "", speaker_name).strip()
        if not cleaned or cleaned.lower() in {"unassigned", "tbd", "unknown", "all", "team", "remote-call"}:
            return None

        users = self.fetch_active_users()
        target = self.speaker_aliases.get(cleaned.lower(), cleaned).lower()

        # Pass 1: Exact match on name, login, or email
        for user in users:
            name = (user.get("name") or "").strip().lower()
            login = (user.get("login") or "").strip().lower()
            email = (user.get("email") or "").strip().lower()
            if target in {name, login, email}:
                return int(user["id"])

        # Pass 2: First-name token or email prefix match
        first_name_matches: list[int] = []
        for user in users:
            name = (user.get("name") or "").strip().lower()
            login_prefix = (user.get("login") or "").split("@")[0].strip().lower()
            name_tokens = name.split()
            if target == login_prefix or (name_tokens and target == name_tokens[0]):
                first_name_matches.append(int(user["id"]))

        if len(first_name_matches) == 1:
            return first_name_matches[0]

        # Pass 3: Fuzzy match across full names and first names
        best_id: int | None = None
        best_score = 0.0
        for user in users:
            name = (user.get("name") or "").strip().lower()
            candidates = [name] + name.split()
            score = max((SequenceMatcher(None, target, cand).ratio() for cand in candidates if cand), default=0.0)
            if score > best_score:
                best_score = score
                best_id = int(user["id"])

        if best_id is not None and best_score >= min_similarity:
            logger.info("Fuzzy-matched speaker '%s' -> res.users ID %s (score=%.2f)", speaker_name, best_id, best_score)
            return best_id

        logger.warning("Could not resolve speaker '%s' to any active Odoo user.", speaker_name)
        return None

    def map_speakers(self, speaker_names: list[str]) -> dict[str, int | None]:
        """Returns a dictionary mapping each unique speaker name to its resolved `res.users` ID."""
        return {name: self.resolve_speaker_to_user_id(name) for name in dict.fromkeys(speaker_names)}

    # -------------------------------------------------------------------------
    # 2. Markdown Action Items Parser & Date Normalizer
    # -------------------------------------------------------------------------

    @staticmethod
    def parse_action_items_from_markdown(markdown_text: str) -> list[ActionItem]:
        """
        Extracts rows from the `## Action Items` Markdown table:
        | Owner | Task | Deadline |
        | :--- | :--- | :--- |
        | Gabriele | Verify inverter telemetry logs | 2026-11-27 |
        """
        section_match = re.search(
            r"##\s*Action Items\s*\n(.*?)(?=\n##\s|\Z)",
            markdown_text,
            flags=re.IGNORECASE | re.DOTALL,
        )
        if not section_match:
            return []

        section_body = section_match.group(1)
        items: list[ActionItem] = []

        for line in section_body.splitlines():
            line = line.strip()
            if not line.startswith("|"):
                continue
            cells = [c.strip() for c in line.strip("|").split("|")]
            if len(cells) < 2:
                continue

            # Skip Markdown header and separator rows (e.g. | :--- | --- |)
            if cells[0].lower() in {"owner", "assignee", "who"} or set(cells[0]) <= {"-", ":"}:
                continue

            owner = re.sub(r"[*`\[\]]", "", cells[0]).strip()
            task = cells[1].strip()
            deadline = cells[2].strip() if len(cells) >= 3 else None

            if task and set(task) > {"-", ":"}:
                items.append(ActionItem(owner=owner, task=task, deadline=deadline))

        return items

    @staticmethod
    def _normalize_deadline(raw_deadline: str | None) -> str | None:
        """Converts YYYY-MM-DD (or DD/MM/YYYY) strings into Odoo `date_deadline` format (YYYY-MM-DD)."""
        if not raw_deadline:
            return None
        cleaned = re.sub(r"[*`]", "", raw_deadline).strip()
        for fmt in ("%Y-%m-%d", "%d/%m/%Y", "%d-%m-%Y"):
            try:
                return datetime.strptime(cleaned, fmt).strftime("%Y-%m-%d")
            except ValueError:
                continue
        return None

    @staticmethod
    def _markdown_to_basic_html(markdown_text: str) -> str:
        """Converts consolidated Markdown minutes into clean HTML for Odoo's `description` field."""
        lines = markdown_text.splitlines()
        html_parts: list[str] = []
        in_list = False

        for line in lines:
            stripped = line.strip()
            if not stripped:
                if in_list:
                    html_parts.append("</ul>")
                    in_list = False
                continue

            if stripped.startswith("## "):
                if in_list:
                    html_parts.append("</ul>")
                    in_list = False
                html_parts.append(f"<h3>{stripped[3:]}</h3>")
            elif stripped.startswith(("- ", "* ")):
                if not in_list:
                    html_parts.append("<ul>")
                    in_list = True
                item_html = re.sub(r"\*\*(.+?)\*\*", r"<strong>\1</strong>", stripped[2:])
                html_parts.append(f"<li>{item_html}</li>")
            else:
                if in_list:
                    html_parts.append("</ul>")
                    in_list = False
                para_html = re.sub(r"\*\*(.+?)\*\*", r"<strong>\1</strong>", stripped)
                html_parts.append(f"<p>{para_html}</p>")

        if in_list:
            html_parts.append("</ul>")

        return "\n".join(html_parts)

    # -------------------------------------------------------------------------
    # 3. Parent Meeting Task + Assigned Sub-Task Creation
    # -------------------------------------------------------------------------

    def publish_meeting_with_subtasks(
        self,
        meeting_id: str,
        consolidated_markdown: str,
        participants: list[str],
        project_id: int | None = None,
    ) -> dict[str, Any]:
        """
        Creates:
          1. A parent `project.task` for the meeting, assigned to all resolved meeting participants.
          2. Individual child `project.task` records (`parent_id = parent_task_id`) for every row
             in the Action Items table, assigned to that specific speaker's `res.users` ID.
        """
        target_project_id = project_id or self.project_id
        speaker_user_map = self.map_speakers(participants)
        participant_user_ids = sorted({uid for uid in speaker_user_map.values() if uid is not None})

        meeting_date = datetime.now().strftime("%Y-%m-%d")
        parent_vals: dict[str, Any] = {
            "name": f"[Meeting Notes] {meeting_id} ({meeting_date})",
            "description": self._markdown_to_basic_html(consolidated_markdown),
        }
        if target_project_id:
            parent_vals["project_id"] = target_project_id
        if participant_user_ids:
            # Odoo 15–18 uses Many2many `user_ids` with Command.set `(6, 0, [...])`
            parent_vals["user_ids"] = [(6, 0, participant_user_ids)]

        parent_task_id = int(self._execute("project.task", "create", [parent_vals]))
        logger.info("Created Odoo parent meeting task ID=%s for '%s'", parent_task_id, meeting_id)

        action_items = self.parse_action_items_from_markdown(consolidated_markdown)
        created_subtasks: list[dict[str, Any]] = []

        for item in action_items:
            owner_uid = self.resolve_speaker_to_user_id(item.owner)
            subtask_vals: dict[str, Any] = {
                "name": item.task,
                "parent_id": parent_task_id,
                "description": (
                    f"<p><strong>Assigned Speaker:</strong> {item.owner}</p>"
                    f"<p><strong>Source Meeting:</strong> {meeting_id} ({meeting_date})</p>"
                ),
            }
            if target_project_id:
                subtask_vals["project_id"] = target_project_id
            if owner_uid is not None:
                subtask_vals["user_ids"] = [(6, 0, [owner_uid])]
            else:
                # Keep owner label visible in title if no matching Odoo user was found
                subtask_vals["name"] = f"[{item.owner}] {item.task}"

            parsed_deadline = self._normalize_deadline(item.deadline)
            if parsed_deadline:
                subtask_vals["date_deadline"] = parsed_deadline

            subtask_id = int(self._execute("project.task", "create", [subtask_vals]))
            created_subtasks.append(
                {
                    "subtask_id": subtask_id,
                    "owner": item.owner,
                    "odoo_user_id": owner_uid,
                    "task": item.task,
                    "deadline": parsed_deadline,
                }
            )

        return {
            "status": "published",
            "parent_task_id": parent_task_id,
            "speaker_user_map": speaker_user_map,
            "subtasks": created_subtasks,
        }

```

### How to Wire It Into `server/hub.py`

In your `/api/meetings/{meeting_id}/finalize` endpoint inside `server/hub.py`, call `publish_meeting_with_subtasks` right after Ollama generates the consolidated Markdown:

```python
from odoo_client import OdooMeetingClient

# Optional alias map for nicknames or middle-name splits in your office
SPEAKER_ALIASES = {
    "Ginadomenico": "Gian Domenico",
    "Muhammad": "muhammad.abbasi.team@gmail.com",
}

odoo = OdooMeetingClient(speaker_aliases=SPEAKER_ALIASES)

# Inside finalize_meeting(meeting_id):
# participants = sorted({seg["speaker"] for seg in finalized_segments})
if odoo.is_configured:
    odoo_result = odoo.publish_meeting_with_subtasks(
        meeting_id=meeting_id,
        consolidated_markdown=summary_md,
        participants=participants,
    )

```

### Key Details Built In

* **Odoo 18 Multi-Assignee & Sub-Task Compatibility:** Uses `user_ids: [(6, 0, [user_id])]` (the Many2many assignee field introduced in Odoo 15+) and `parent_id: parent_task_id` so every action item appears natively under the **Sub-tasks** tab of the main meeting task and shows up in each colleague's **My Tasks** board.
* **4-Tier Speaker Resolution:** Resolves workstation tags like `Gabriele` or `Ginadomenico` against active internal `res.users` records via explicit aliases, exact name/login match, unique first-name/email-prefix match, and fuzzy string similarity (`>= 0.78`) before falling back to prefixing `[Owner]` in the sub-task title if a guest speaker has no Odoo account.
* **Automatic Deadline Parsing:** Normalizes `YYYY-MM-DD` or `DD/MM/YYYY` strings from Ollama's Markdown table into Odoo's `date_deadline` field.
